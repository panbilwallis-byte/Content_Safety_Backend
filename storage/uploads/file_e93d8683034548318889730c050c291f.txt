最近在刷华为题遇到了一道本题的该题，题目是一样的，但多了一个要求，二叉搜索树的高度需要小于等于k。即要求节点数为n，且高度小于等于k的二叉搜索树的数量。处理就需要用到记忆化搜索和递归了，这里给出程序，大家感兴趣可以学习一下
class Solution
{
private:
    vector<vector<long long>> dp;
public:
    long long helper(int n, int k)
    {
        dp[n][k] = 0;
        
        if (pow(2,k) <= n)
            return 0;

        for (int i = 0; i < n; i++)
        {
            long long a, b;
            if (dp[i][k - 1] >= 0)
                a = dp[i][k - 1];
            else
                a = helper(i, k - 1);

            if (dp[n - i - 1][k - 1] >= 0)
                b = dp[n - i - 1][k - 1];
            else
                b = helper(n - i - 1, k - 1);

            dp[n][k] += a * b;
        }

        return dp[n][k];
    }

    int SearchTreeNum(int n, int k)
    {
        dp = vector<vector<long long>>(n + 1, vector<long long>(k + 1, -1));
        for (int i = 0; i <= k; i++)
            dp[0][i] = 1;

        return helper(n, k);
    }
};



int main()
{
    Solution st;
    
    int n, k;
    cin >> n >> k;

    long long res = st.SearchTreeNum(n,k);
    printf("%lld\n", res);

    return 0;
}
